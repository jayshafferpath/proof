25+export function createCouchdropClient(config: CouchdropClientConfig): CouchdropClient {
26+  if (config.apiBaseUrl.trim() === "") {
27+    throw new CouchdropConfigError("createCouchdropClient: config.apiBaseUrl must be non-empty.");
28+  }
29+  if (config.fileioBaseUrl.trim() === "") {
30+    throw new CouchdropConfigError(
31+      "createCouchdropClient: config.fileioBaseUrl must be non-empty.",
32+    );
33+  }
34+  if (config.apiToken.trim() === "") {
35+    throw new CouchdropConfigError("createCouchdropClient: config.apiToken must be non-empty.");
36+  }
37+
38+  const apiBaseUrl = trimTrailingSlash(config.apiBaseUrl);
39+  const fileioBaseUrl = trimTrailingSlash(config.fileioBaseUrl);
40+  const fetchImpl = config.fetchImpl ?? fetch;
41+
42+  const uploadBytes = async (input: UploadRecipientScopedFileInput, remotePath: string) => {
43+    const query = new URLSearchParams({ path: remotePath, size: String(input.sizeBytes) });
44+    let response: Response;
45+    try {
46+      response = await fetchImpl(`${fileioBaseUrl}${UPLOAD_PATH}?${query.toString()}`, {
47+        method: "POST",
48+        headers: { token: config.apiToken, "content-type": input.mimeType },
49+        body: input.contentStream as unknown as RequestInit["body"],
50+        duplex: "half",
51+      } as RequestInit & { duplex: "half" });
52+    } catch (cause) {
53+      throw new CouchdropUploadError(
54+        `Couchdrop upload failed for ${input.filename}: request could not be completed.`,
55+        { cause },
56+      );
57+    }
58+    if (!response.ok) {
59+      throw new CouchdropUploadError(
60+        `Couchdrop upload failed for ${input.filename}: HTTP ${response.status}.`,
61+        { status: response.status },
62+      );
63+    }
64+  };
65+
66+  const createSharedLink = async (
67+    input: UploadRecipientScopedFileInput,
68+    remotePath: string,
69+  ): Promise<SharedLinkResponse> => {
70+    let response: Response;
71+    try {
72+      response = await fetchImpl(`${apiBaseUrl}${SHARED_LINKS_PATH}`, {
73+        method: "PUT",
74+        headers: { token: config.apiToken, "content-type": "application/json" },
75+        // TODO(NEV-1643): confirm the recipient-email scoping field against the
76+        // live tenant; the public docs expose password/expiry but not email scoping.
77+        body: JSON.stringify({ path: remotePath, restrictEmail: input.recipientEmail }),
78+      });
79+    } catch (cause) {
80+      throw new CouchdropUploadError(
81+        `Couchdrop shared-link creation failed for ${input.filename}: request could not be completed.`,
82+        { cause },
83+      );
84+    }
85+    if (!response.ok) {
86+      throw new CouchdropUploadError(
87+        `Couchdrop shared-link creation failed for ${input.filename}: HTTP ${response.status}.`,
88+        { status: response.status },
89+      );
90+    }
91+    try {
92+      return (await response.json()) as SharedLinkResponse;
93+    } catch (cause) {
94+      throw new CouchdropUploadError(
95+        `Couchdrop shared-link creation for ${input.filename} returned an unreadable response body.`,
96+        { cause, status: response.status },
97+      );
98+    }
99+  };
100+
101+  const setExpiry = async (id: string, remotePath: string, linkTtlSeconds: number) => {
102+    const expiry = String(Math.floor(Date.now() / 1000) + linkTtlSeconds);
103+    const response = await fetchImpl(`${apiBaseUrl}${SHARED_LINKS_PATH}`, {
104+      method: "POST",
105+      headers: { token: config.apiToken, "content-type": "application/json" },
106+      body: JSON.stringify({ id, path: remotePath, expiry }),
107+    }).catch((cause) => {
108+      throw new CouchdropUploadError(
109+        "Couchdrop shared-link expiry update failed: request could not be completed.",
110+        { cause },
111+      );
112+    });
113+    if (!response.ok) {
114+      throw new CouchdropUploadError(
115+        `Couchdrop shared-link expiry update failed: HTTP ${response.status}.`,
116+        { status: response.status },
117+      );
118+    }
119+  };
120+
121+  return {
122+    async uploadRecipientScopedFile(
123+      input: UploadRecipientScopedFileInput,
124+    ): Promise<UploadRecipientScopedFileResult> {
125+      const remotePath = buildRemotePath(input);
126+
127+      await uploadBytes(input, remotePath);
128+      const link = await createSharedLink(input, remotePath);
129+
130+      if (link.id == null) {
131+        throw new CouchdropUploadError(
132+          `Couchdrop shared-link creation for ${input.filename} returned no link id.`,
133+        );
134+      }
135+
136+      if (input.linkTtlSeconds !== undefined) {
137+        await setExpiry(link.id, remotePath, input.linkTtlSeconds);
138+      }
139+
140+      // TODO(NEV-1643): confirm the shareable-URL field on the live response; the
141+      // documented create response exposes id/path but does not name a URL field.
142+      const sharedLinkUrl = link.url ?? link.link ?? `${apiBaseUrl}${SHARED_LINKS_PATH}/${link.id}`;
143+
144+      return {
145+        sharedLinkUrl,
146+        uploadId: link.id,
147+        remotePath: link.path ?? remotePath,
148+      };
149+    },
150+  };
